// ─── Build Columns from selectedColumns ───
 
function buildColumns(
  selectedColumns: string[],
  onRowClick: (assetId: string) => void,
): ColumnsType<Asset> {
  return selectedColumns
    .filter((col) => COLUMN_BUILDERS[col])
    .map((col) => COLUMN_BUILDERS[col](onRowClick));
}
