func (uc *Usecase) submitItem(ctx, job, item, templateVersion) (submitOutcome, error) {
    runID := item.PipelineRunID   // crash-resume 入口
    workflowName := item.WorkflowName

    // ① 已经存在 runID 时: 检查 Argo 实际是否跑起来了
    if runID != "" {
        existing, getErr := uc.deployer.GetRun(ctx, runID)
        if getErr == nil && runAlreadySubmitted(existing) {
            wf := existing.WorkflowName  // 如果为空退回到 item 上的
            return outcomeOK,
                uc.repo.UpdateItemPipelineRun(ctx, item.ID, runID, wf, "submitted")
        }
    }

    // ② 没有 runID: 落 ledger 行 + 拿 deterministic name
    if runID == "" {
        runID, workflowName, initErr = uc.deployer.UpsertBatchSubtaskRun(...)
        // ...
        uc.repo.UpdateItemPipelineRun(ctx, item.ID, runID, workflowName, "pending")
    }

    dep, err := uc.deployer.DeployByTemplateID(ctx, job.TemplateID, "", []string{item.AssetID}, deployOpts)
    if err != nil {
        switch {                                            // 错误洋葱
        case isWorkflowAlreadyExists(err):
            // AlreadyExists: 回填 UID (crash-resume)
            refreshed, _ := uc.deployer.RefreshRunFromWorkflowByName(ctx, workflowName, "")
            if refreshed == nil || !runAlreadySubmitted(refreshed) {
                // TTL 清理后 CR 不可读 - 标 submitted, 让 watcher 兜底
                return outcomeOK, uc.repo.UpdateItemPipelineRun(...)
            }
            return outcomeOK, uc.repo.UpdateItemPipelineRun(...)
        case errors.Is(err, pipelineUC.ErrWorkflowSubmitIncomplete):
            // 不当失败: 下 cycle 通过 AlreadyExists 自愈
            return outcomeTransient, err
        case classifySubmitError(err) == outcomeTransient:
            attempts, _ := uc.repo.IncrementItemSubmitAttempts(ctx, item.ID)
            if attempts < maxSubmitAttempts { return outcomeTransient, err }
            // poison: 进 DLQ
            metrics.DispatcherDLQTotal.WithLabelValues("max_submit_attempts").Inc()
            return outcomePermanent, uc.repo.UpdateItemPipelineRun(... "failed")
        default:
            // permanent (transpile / template / asset)
            metrics.DispatcherDLQTotal.WithLabelValues("permanent").Inc()
            return outcomePermanent, uc.deployer.RecordBatchSubtaskFailure(...)
        }
    }
    return outcomeOK, uc.repo.UpdateItemPipelineRun(... "submitted")
}

-- item-side WHERE gates on "currently NON-terminal" — this is the
-- single source of idempotency. A redelivered webhook finds the row
-- already terminal, matches 0 rows in `upd`, and the job-side UPDATE
-- runs against an empty IN() set so the counter is not touched.
WITH upd AS (
  UPDATE backfill_items
  SET status = $2,
      workflow_name = COALESCE(NULLIF($3, ''), workflow_name),
      error_message = CASE WHEN $2 = 'completed' THEN '' ELSE COALESCE(NULLIF($4, ''), error_message) END,
      finished_at = NOW(),
      started_at  = COALESCE(started_at, NOW())
  WHERE id = $1 AND status NOT IN ('completed', 'failed', 'cancelled')
  RETURNING job_id
)
UPDATE backfill_jobs
SET completed_count = completed_count + (CASE WHEN $2 = 'completed' THEN 1 ELSE 0 END),
    failed_count    = failed_count    + (CASE WHEN $2 IN ('failed', 'cancelled') THEN 1 ELSE 0 END),
    updated_at = NOW()
WHERE id IN (SELECT job_id FROM upd)

完整 PG 序列化确保两条 UPDATE 在同一 transaction 里 → item 推进和 counter bump 要么都发生，要么都失败。普通 IncrementCompleted 没有这个守门（会重复 +1），所以这是新方法而不是 wrapper。

func (uc *Usecase) bulkSyncClusterRuns(ctx, runs, group, residualLimit, recalibrate) int {
    // 1. resolve cluster clientset
    client, err := uc.resolveArgoClientForRun(ctx, &runs[group[0]])

    // 2. 1× ListWorkflows per namespace in group (几乎总是 1 个)
    namespaces := map[string]bool{}
    for _, i := range group { namespaces[runs[i].ArgoNamespace] = true }
    listed := map[string]*wfv1.Workflow{}
    listOK := true
    for ns := range namespaces {
        items, err := client.ListWorkflows(ctx, ns, activeWorkflowSelector)
        if err != nil {
            listOK = false ; break   // 失败 → 整 tick 降级
        }
        for i := range items { listed[ns+"/"+items[i].Name] = &items[i] }
    }
    if !listOK { return uc.legacyRefreshGroup(...) }   // bounded per-run GET

    // 3. apply matches (RV gate)
    synced, residuals := 0, 0
    for _, i := range group {
        run := &runs[i]
        if wf, ok := listed[ns+"/+strings.TrimSpace(run.WorkflowName)]; ok {
            if recalibrate || uc.markWorkflowApplied(run.ID, wf.ResourceVersion) {
                uc.applyWorkflowToRun(ctx, run, wf, nodeProjectTerminalArchive)
                if !isActiveDeploymentStatus(run.Status) { uc.forgetWorkflowApplied(run.ID) }
                synced++
            }
            continue
        }
        // 4. absentees → bounded targeted GET
        if residuals < residualLimit {
            uc.refreshPipelineRunStatus(ctx, run)
            uc.forgetWorkflowApplied(run.ID)
            residuals++
            synced++
        }
    }
    return synced
}

60s 自愈的逻辑链

backend/internal/usecase/backfill/usecase.go:StartJobReconciler · syncJobProgressForce

webhook 是 push（O(1)），reconciler 是 pull（全扫 + 覆写）。两者职责不重叠：

  • webhook 负责 counter 立即推进（≤1s 延迟）
  • reconciler 负责 counter 始终来自权威（防止 drift 累积）
  • counter 永远 = SummarizeItemStatuses(jobs).{completed,failed} 在每个 60s 节点

这就是为什么 webhook 的 +=1 + reconciler 的 = 配合得很好 — 像一个 CRDT 的"逻辑时钟 + 校准基线"组合。

// 每 cycle endCycle 调一次
func (g *clusterGovernor) endCycle() {
    g.mu.Lock()
    defer g.mu.Unlock()
    if g.total == 0 { return }
    distress := g.throttled || g.slow*2 > g.total
    if distress {
        g.effective /= 2
        if g.effective < g.floor { g.effective = g.floor }
    } else {
        g.effective = g.effective * 3 / 2
        if g.effective > g.configured { g.effective = g.configured }
    }
    g.slow, g.total, g.throttled = 0, 0, false
}

AIMD: Additive Increase / Multiplicative Decrease。经典拥塞控制（TCP 风格）。effective 是 in-flight concurrency（不是 rate），与 token bucket 解耦。

  • floor = configured / 4（最少保 25% 带宽）
  • configured 来自 CYB-3679 在线调优表
  • healthy 时 ×1.5（不是 +1 — AIMD 不该过分冒险）
